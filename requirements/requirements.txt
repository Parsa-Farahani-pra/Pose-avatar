# available in 0.10.14 but removed in later 0.10.x patches (0.10.20+) and in

# the solutions subpackage entirely, so we pin to the last working version.

# opencv-contrib-python provides cv2 (imshow + MediaPipe's extra modules).
opencv-contrib-python>=4.8.0,<5
numpy>=1.24.0
# Ursina 7.x is the 3D engine used by pose_avatar.py (Panda3D based).
ursina>=7.0.0,<8
